# 0. 出発の補正 v0: 今の既定の補正を 84 個のデッキの語彙に広げ（476 枚）、新しいカードの行を属性で埋めたもの。
#    台帳の「新しいプール-属性で埋めた補正」と同じ作り方で、今の既定からなら指紋 aa2d1cc1（小さく通した回で同じになった）。
#    足した入力（myDeckCards・oppUnseen）の行は 0 のまま（属性で埋めるのは、学んだ行があるゾーンだけ）
td extend --model data/value-net.json --vocab @data/pools/azvaldt/sets/all84.txt --out @/v0e.json --est 10
td attr --model @/v0e.json --out @/v0.json --keepA 0 --est 30
# v0 の側の見込みは、今の既定のまま（写すだけ。R2 からは前の回の見込みにする）
td extend --model data/potential-net.json --out @/v0pot.json --est 10

# 1. 自己対戦 6 万試合（g7000000〜7059999）。3000 試合ずつ s-1〜s-20。B・relearn9 と同じ打ち方で、デッキだけ 84 個（全部の組を 8 回ほど回る）。
#    見込みは v0 の側のもの（R1 は今の既定を写したもので、渡さないときと同じ試合）
td selfplay --model @/v0.json --decks @data/pools/azvaldt/sets/all84.txt --games 60000 --start 7000000 --chunk 3000 --eps 0.1 --opp beam --sampling default --iv 3 --ivZone both --imagined 2 --searchValue 1 --potential @/v0pot.json --out @/s-{k}.jsonl.gz --est 0.40

# 2. 学習（B と同じ学び方）、間引き、属性で埋める → r.json
td train --impl rust --model @/v0.json --data @/s-{1..20}.jsonl.gz --out @/rt.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --dropInputs myDeckCards,oppUnseen --est 900
td prune --impl rust --model @/rt.json --data @/s-{1..20}.jsonl.gz --out @/rp.json --est 120
td attr --model @/rp.json --out @/r.json --keepA 0 --est 60

# 3. 見込みを、学んだ補正の点数で学び直す（beam-gap 2000 試合、84 個のデッキ、広げた語彙。語彙は r.json から）
beam-gap --games 2000 --from 1000 --chunk 500 --decks @data/pools/azvaldt/sets/all84.txt --vocab @/r.json --ref '{"beamWidth":4096,"potentialNet":null,"valueNet":"@/r.json"}' --variants '{"w32":{"beamWidth":32,"potentialNet":null,"valueNet":"@/r.json"}}' --sample 100 --inputs @/pot-{k}.jsonl --out @/pot-{k}-rows.jsonl --est 0.40
potential train --data @/pot-{1..4}.jsonl --valid 2800-2999 --out @/rpot.json --est 300

# 4. S1: R1 対 v0（84 個のデッキの鏡の compare、4000 試合）。v0 の側の見込みは今の既定のまま、学んだ側は学び直した見込み
compare 2000 '{"valueNet":"@/r.json","potentialNet":"@/rpot.json"}' '{"valueNet":"@/v0.json","potentialNet":"@/v0pot.json"}' 0 --decks @data/pools/azvaldt/sets/all84.txt --label 新しいプール-R1 --est 0.16

# 5. 線ではないが毎回見る数字: PP とプレイできるカードを残して終えたターンの割合（今の補正 29%、v0 15%、スターター 6%）・1 試合のターンの数・型ごとの勝率。
