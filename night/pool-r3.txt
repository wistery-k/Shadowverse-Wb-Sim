# 0. 出発の補正: R2 の補正と見込みを写す（R4 なら前の回のフォルダに替える）
td extend --model /home/wk/svwb-data/night/pool-r2/r.json --out @/v0.json --est 10
td extend --model /home/wk/svwb-data/night/pool-r2/rpot.json --out @/v0pot.json --est 10

# 1. 自己対戦 6 万試合（g7200000〜7259999）。3000 試合ずつ s-1〜s-20。R1・R2 と同じ打ち方。見込みは v0 の側のもの（R2 の学び直した見込みの写し）
td selfplay --model @/v0.json --decks @data/pools/azvaldt/sets/all84.txt --games 60000 --start 7200000 --chunk 3000 --eps 0.1 --opp beam --sampling default --iv 3 --ivZone both --imagined 2 --searchValue 1 --potential @/v0pot.json --out @/s-{k}.jsonl.gz --est 0.50

# 2. 学習（R1・R2 と同じ学び方）、間引き、属性で埋める → r.json
td train --impl rust --model @/v0.json --data @/s-{1..20}.jsonl.gz --out @/rt.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --est 1200
td prune --impl rust --model @/rt.json --data @/s-{1..20}.jsonl.gz --out @/rp.json --est 90
td attr --model @/rp.json --out @/r.json --keepA 0 --est 60

# 3. 見込みを、学んだ補正の点数で学び直す（R1 と同じ。語彙は r.json から）
beam-gap --games 2000 --from 1000 --chunk 500 --decks @data/pools/azvaldt/sets/all84.txt --vocab @/r.json --ref '{"beamWidth":4096,"potentialNet":null,"valueNet":"@/r.json"}' --variants '{"w32":{"beamWidth":32,"potentialNet":null,"valueNet":"@/r.json"}}' --sample 100 --inputs @/pot-{k}.jsonl --out @/pot-{k}-rows.jsonl --est 0.36
potential train --data @/pot-{1..4}.jsonl --valid 2800-2999 --out @/rpot.json --est 180

# 4. S2 の形: R3 対 R2（84 個のデッキの鏡の compare、1.2 万試合）。R2 の側は R2 の補正と学び直した見込み、R3 の側は R3 の補正と学び直した見込み
compare 6000 '{"valueNet":"@/r.json","potentialNet":"@/rpot.json"}' '{"valueNet":"@/v0.json","potentialNet":"@/v0pot.json"}' 0 --decks @data/pools/azvaldt/sets/all84.txt --label 新しいプール-R3 --est 0.17
