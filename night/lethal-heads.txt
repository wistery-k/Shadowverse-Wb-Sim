# 1. 自己対戦 6 万試合（g7400000〜7459999。R4 は 7300000〜）。R4 と同じ打ち方に、ターン末ごとの相手のリーサルの印（--lethalLabel 1）を足す
td selfplay --model rust/engine/yardstick/pool-r4-net.json --decks @data/pools/azvaldt/sets/all84.txt --games 60000 --start 7400000 --chunk 3000 --eps 0.1 --opp beam --sampling default --iv 3 --ivZone both --imagined 2 --searchValue 1 --lethalLabel 1 --potential rust/engine/yardstick/pool-r4-potential.json --out @/s-{k}.jsonl.gz --est 0.58

# 2. C: 今までと同じ学び方（R1〜R4 と同じ）、間引き、属性で埋める
td train --impl rust --model rust/engine/yardstick/pool-r4-net.json --data @/s-{1..20}.jsonl.gz --out @/ct.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --est 1200
td prune --impl rust --model @/ct.json --data @/s-{1..20}.jsonl.gz --out @/cp.json --est 90
td attr --model @/cp.json --out @/c.json --keepA 0 --est 60

# 2'. X: リーサルの出力を補助に足して一緒に学ぶ（勝つ確率は C と同じ形）
td train --impl rust --model rust/engine/yardstick/pool-r4-net.json --data @/s-{1..20}.jsonl.gz --out @/xt.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --lethalHead aux --est 1300
td prune --impl rust --model @/xt.json --data @/s-{1..20}.jsonl.gz --out @/xp.json --est 90
td attr --model @/xp.json --out @/x.json --keepA 0 --est 60

# 2''. Z: 勝つ確率を (1 − リーサルの出力) × 生き延びたときの勝率 にして学ぶ
td train --impl rust --model rust/engine/yardstick/pool-r4-net.json --data @/s-{1..20}.jsonl.gz --out @/zt.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --lethalHead survival --est 1300
td prune --impl rust --model @/zt.json --data @/s-{1..20}.jsonl.gz --out @/zp.json --est 90
td attr --model @/zp.json --out @/z.json --keepA 0 --est 60

# 3. 線ではないが毎回見る数字（C。R1〜R4 と同じ）
profile --opts '{"valueNet":"@/c.json","potentialNet":"rust/engine/yardstick/pool-r4-potential.json"}' --games 420 --decks @data/pools/azvaldt/sets/all84.txt --est 0.16
