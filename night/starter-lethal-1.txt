# 回の名前は starter-lethal-1（2 晩目のキューがこの回のフォルダのモデルを読む）

# 1. 自己対戦 6.3 万試合（g6400000〜6462999。前の学び直しの g6300000〜6359999・g6390000〜 と重ならない）。3000 試合ずつ s-1〜s-21。
#    今の既定と同じ打ち方（relearn-b と同じ引数）に、ターン末ごとの次の相手のターンに負けたかの印（--lostLabel 1）を足す。s-21 は学びに使わない（翌朝の取り置き）
td selfplay --model data/value-net.json --games 63000 --start 6400000 --chunk 3000 --eps 0.1 --opp beam --sampling default --iv 3 --ivZone both --imagined 2 --searchValue 1 --lostLabel 1 --out @/s-{k}.jsonl.gz --est 0.42

# 2. C': 今の形のまま、今の既定から学び直す（新しいプールの C と同じ学び方）→ 間引き → 属性で埋める
td train --impl rust --model data/value-net.json --data @/s-{1..20}.jsonl.gz --out @/ct.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --est 900
td prune --impl rust --model @/ct.json --data @/s-{1..20}.jsonl.gz --out @/cp.json --est 120
td attr --model @/cp.json --out @/c.json --keepA 0 --est 60

# 2'. Z': 勝つ確率を (1 − リーサルの出力) × 生き延びたときの勝率 にして学ぶ（新しいプールの Z と同じ）
td train --impl rust --model data/value-net.json --data @/s-{1..20}.jsonl.gz --out @/zt.json --epochs 3 --lr 0.0003 --l2 0.001 --best 1 --bestOn end,imagined --ivRes 0 --base zero --lethalHead survival --est 1000
td prune --impl rust --model @/zt.json --data @/s-{1..20}.jsonl.gz --out @/zp.json --est 120
td attr --model @/zp.json --out @/z.json --keepA 0 --est 60
